644 private void AddNextCell(Cell cell, bool incremental, Style style)
645 {
646 // date and time styles are already defined by the passed cell object
647 if (style != null || (activeStyle != null && useActiveStyle))
648 {
649 if (cell.CellStyle == null && useActiveStyle)
650 {
651 cell.SetStyle(activeStyle);
652 }
653 else if (cell.CellStyle == null && style != null)
654 {
655 cell.SetStyle(style);
656 }
657 else if (cell.CellStyle != null && useActiveStyle)
658 {
659 Style mixedStyle = (Style)cell.CellStyle.Copy();
660 mixedStyle.Append(activeStyle);
661 cell.SetStyle(mixedStyle);
662 }
663 else if (cell.CellStyle != null && style != null)
664 {
665 Style mixedStyle = (Style)cell.CellStyle.Copy();
666 mixedStyle.Append(style);
667 cell.SetStyle(mixedStyle);
668 }
669 }
670 CellKey cellKey = new CellKey(cell.ColumnNumber, cell.RowNumber);
671 if (cells.TryGetValue(cellKey, out Cell previousCell))
672 {
673 previousCell.UnbindFeatures();
674 }
675 cells[cellKey] = cell;
676 cell.BindFeatures(Features);
677 if (incremental)
678 {
679 if (CurrentCellDirection == CellDirection.ColumnToColumn)
680 {
681 currentColumnNumber++;
682 }
683 else if (CurrentCellDirection == CellDirection.RowToRow)
684 {
685 currentRowNumber++;
686 }
687 else
688 {
689 // disabled / no-op
690 }
691 }
692 else
693 {
694 if (CurrentCellDirection == CellDirection.ColumnToColumn)
695 {
696 currentColumnNumber = cell.ColumnNumber + 1;
697 currentRowNumber = cell.RowNumber;
698 }
699 else if (CurrentCellDirection == CellDirection.RowToRow)
700 {
701 currentColumnNumber = cell.ColumnNumber;
702 currentRowNumber = cell.RowNumber + 1;
703 }
704 else
705 {
706 // disabled / no-op
707 }
708 }
709 }

1029 internal IReadOnlyList<Address> AddDefinedNameArrayCells(Cell masterCell, Range arrayRange, Style style)
1030 {
1031 IReadOnlyList<Address> addresses = arrayRange.ResolveEnclosedAddresses();
1032 List<Address> addedAddresses = new List<Address>();
1033 foreach (Address address in addresses)
1034 {
1035 if (address.Row == masterCell.RowNumber && address.Column == masterCell.ColumnNumber)
1036 {
1037 continue; // Skip master cell
1038 }
1039 Cell arrayRefCell = new Cell(null, Cell.CellType.Formula);
1040 arrayRefCell.Formula.MasterCellAddress = masterCell.CellAddress;
1041 arrayRefCell.Formula.FormulaRange = masterCell.Formula.FormulaRange;
1042 arrayRefCell.Formula.Type = FormulaData.FormulaType.Array;
1043 AddCell(arrayRefCell, address.Column, address.Row, style);
1044 addedAddresses.Add(address);
1045 }
1046 return addedAddresses;
1047 }

1160 private void AddCellRangeInternal<T>(IReadOnlyList<T> values, Address startAddress, Address endAddress, Style style)
1161 {
1162 if (values == null)
1163 {
1164 throw new RangeException("The passed value list cannot be null");
1165 }
1166 List<Address> addresses = Cell.GetCellRange(startAddress, endAddress) as List<Address>;
1167 if (values.Count != addresses.Count)
1168 {
1169 throw new RangeException("The number of passed values (" + values.Count + ") differs from the number of cells within the range (" + addresses.Count + ")");
1170 }
1171 List<Cell> list = Cell.ConvertArray(values) as List<Cell>;
1172 int len = values.Count;
1173 for (int i = 0; i < len; i++)
1174 {
1175 list[i].RowNumber = addresses[i].Row;
1176 list[i].ColumnNumber = addresses[i].Column;
1177 AddNextCell(list[i], false, style);
1178 }
1179 }

1476 private int GetBoundaryDataNumber(bool row, bool min, bool ignoreEmpty)
1477 {
1478 if (cells.Count == 0)
1479 {
1480 return -1;
1481 }
1482 if (!ignoreEmpty)
1483 {
1484 if (row && min)
1485 {
1486 return cells.Values.Min(x => x.RowNumber);
1487 }
1488 else if (row)
1489 {
1490 return cells.Values.Max(x => x.RowNumber);
1491 }

2949 if (match.Groups.Count > 1)
