196 public static string GenerateLegacyPasswordHash(string password)
197 {
198 if (string.IsNullOrEmpty(password)) { return string.Empty; }
199 int passwordLength = password.Length;
200 int passwordHash = 0;
201 for (int i = passwordLength; i > 0; i--)
202 {
203 char character = password[i - 1];
204 passwordHash = ((passwordHash >> 14) & 0x01)
205 | ((passwordHash << 1) & 0x7fff);
206 passwordHash ^= character;
207 }
208 passwordHash = ((passwordHash >> 14) & 0x01)
209 | ((passwordHash << 1) & 0x7fff);
210 passwordHash ^= (0x8000 | ('N' << 8) | 'K');
211 passwordHash ^= passwordLength;
212 return passwordHash.ToString("X", ParserUtils.InvariantCulture);
213 }

220 private static SecureString GetSecureString(string plaintextPassword)
221 {
222 SecureString result = new SecureString();
223 try
224 {
225 if (!string.IsNullOrEmpty(plaintextPassword))
226 {
227 foreach (char character in plaintextPassword)
228 {
229 result.AppendChar(character);
230 }
231 }
232 return result;
233 }

236 result.Dispose();

246 private static string GetPasswordOfSecureString(SecureString secureString)
247 {
248 IntPtr unmanagedString = IntPtr.Zero;
249 try
250 {
251 unmanagedString = Marshal.SecureStringToGlobalAllocUnicode(secureString);
252 // Preserve the snapshot's null-terminated string conversion behavior.
253 return Marshal.PtrToStringUni(unmanagedString);
254 }
255 finally
256 {
257 if (unmanagedString != IntPtr.Zero)
258 {
259 Marshal.ZeroFreeGlobalAllocUnicode(unmanagedString);
260 }
261 }
