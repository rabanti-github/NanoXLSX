58 public static void ValidateCellAddressExpression(string expression, Cell.AddressScope scope = Cell.AddressScope.Any)
59 {
60 bool isCellAddress = false;
61 bool isRange = false;
62 Exception lastException = null;
63 try
64 {
65 Cell.ResolveCellCoordinate(expression);
66 isCellAddress = true;
67 }
68 catch (Exception ex)
69 {
70 if (scope == Cell.AddressScope.SingleAddress)
71 {
72 throw new FormatException(ex.Message, ex); // No further checks necessary
73 }
74 lastException = ex;
75 }
76 try
77 {
78 Cell.ResolveCellRange(expression);
79 isRange = true;
80 }
81 catch (Exception ex)
82 {
83 if (scope == Cell.AddressScope.Range)
84 {
85 throw new FormatException(ex.Message, ex); // No further checks necessary
86 }
87 lastException = ex;
88 }
89 if (scope == Cell.AddressScope.Range && isCellAddress && isRange)
90 {
91 System.FormatException innerException = new System.FormatException("The expression (" + expression + ") is a single cell address, but a cell range was expected");
92 throw new FormatException(innerException.Message, innerException);
93 }
94 else if (scope == Cell.AddressScope.Any && !isCellAddress && !isRange)
95 {
96 throw new FormatException(lastException.Message, lastException); // Not a cell or range
97 }
98 else if (scope == Cell.AddressScope.Invalid && (isCellAddress || isRange))
99 {
100 throw new FormatException("The passed expression is valid cell address or range, but the validation was explicitly inverted");
101 }
102 }

134 private static string ValidateColorInternal(string hexCode, bool useAlpha, bool allowEmpty)
135 {
136 if (string.IsNullOrEmpty(hexCode))
137 {
138 if (allowEmpty)
139 {
140 return null;
141 }
142 return "The color expression cannot be null or empty";
143 }
144
145 int length = useAlpha ? 8 : 6;
146 if (hexCode.Length != length)
147 {
148 return "The value '" + hexCode + "' is invalid. A valid value must contain " + length + " hex characters";
149 }
150 if (!Regex.IsMatch(hexCode, "[a-fA-F0-9]{6,8}"))
151 {
152 return "The expression '" + hexCode + "' is not a valid hex value";
153 }
154 return null;
155 }
