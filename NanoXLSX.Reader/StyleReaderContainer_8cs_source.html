53 public void AddStyleComponent(AbstractStyle component)
54 {
55 Type t = component.GetType();
56 if (t == typeof(CellXf))
57 {
58 cellXfs.Add(component as CellXf);
59 }
60 else if (t == typeof(NumberFormat))
61 {
62 NumberFormat nf = component as NumberFormat;
63 if (nf.InternalID.HasValue && !numberFormats.ContainsKey(nf.InternalID.Value))
64 {
65 numberFormats.Add(nf.InternalID.Value, nf);
66 }
67 }
68 else if (t == typeof(Style))
69 {
70 styles.Add(component as Style);
71 }
72 else if (t == typeof(Border))
73 {
74 borders.Add(component as Border);
75 }
76 else if (t == typeof(Fill))
77 {
78 fills.Add(component as Fill);
79 }
80 else if (t == typeof(Font))
81 {
82 fonts.Add(component as Font);
83 }
84 }

208 private AbstractStyle GetComponent(Type type, int index)
209 {
210 try
211 {
212 if (type == typeof(NumberFormat))
213 {
214 NumberFormat numberFormat;
215 if (numberFormats.TryGetValue(index, out numberFormat))
216 {
217 return numberFormat;
218 }
219 throw new StyleException("The number format with the numFmtId: " + index + " was not found");
220 }
221 else if (type == typeof(Style))
222 {
223 return styles[index];
224 }
225 else if (type == typeof(Border))
226 {
227 return borders[index];
228 }
229 else if (type == typeof(Fill))
230 {
231 return fills[index];
232 }
233 else // must be font (CellXF is not handled here)
234 {
235 return fonts[index];
236 }
237 }
238 catch (Exception)
239 {
240 // Ignore
241 }
242 return null;
243 }
